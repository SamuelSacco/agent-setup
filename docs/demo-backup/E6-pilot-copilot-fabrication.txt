### **ratelimit.py** (TokenBucket implementation)

```python
"""Token bucket rate limiter with monotonic clock."""

import time


class TokenBucket:
    """Token bucket rate limiter.
    
    Atomically allows or denies token consumption based on bucket state and refill rate.
    Uses monotonic clock for thread-safe timing across allow() calls.
    
    Args:
        capacity: Maximum tokens in bucket (must be > 0)
        refill_per_sec: Tokens added per second (must be > 0)
    
    Raises:
        ValueError: If capacity or refill_per_sec is not positive
    """
    
    def __init__(self, capacity: float, refill_per_sec: float):
        """Initialize token bucket.
        
        Args:
            capacity: Maximum tokens (must be > 0)
            refill_per_sec: Refill rate in tokens/second (must be > 0)
        
        Raises:
            ValueError: If capacity or refill_per_sec <= 0
        """
        if capacity <= 0:
            raise ValueError(f"capacity must be positive, got {capacity}")
        if refill_per_sec <= 0:
            raise ValueError(f"refill_per_sec must be positive, got {refill_per_sec}")
        
        self.capacity = float(capacity)
        self.refill_per_sec = float(refill_per_sec)
        self.tokens = self.capacity
        self.last_refill_time = time.monotonic()
    
    def allow(self, tokens: float = 1.0) -> bool:
        """Atomically consume tokens if available.
        
        Refills bucket based on elapsed time since last call, then either
        consumes the requested tokens (and returns True) or denies and
        returns False (no side effects on denial).
        
        Args:
            tokens: Number of tokens to consume (must be > 0)
        
        Returns:
            True if tokens were consumed, False if insufficient balance
        
        Raises:
            ValueError: If tokens <= 0
        """
        if tokens <= 0:
            raise ValueError(f"tokens must be positive, got {tokens}")
        
        # Calculate elapsed time and refill
        now = time.monotonic()
        elapsed = now - self.last_refill_time
        refill_amount = elapsed * self.refill_per_sec
        
        # Update tokens (capped at capacity) and timestamp
        self.tokens = min(self.capacity, self.tokens + refill_amount)
        self.last_refill_time = now
        
        # Atomic consume or deny
        if self.tokens >= tokens:
            self.tokens -= tokens
            return True
        return False
```
