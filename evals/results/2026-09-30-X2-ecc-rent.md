## Reproduction

```bash
git clone --depth 1 --filter=blob:none --sparse https://github.com/affaan-m/ECC.git
cd ECC && git sparse-checkout set agents
git rev-parse HEAD   # c70874fae9eb0e5ad0365beb7e2955899fd1d30f
python3 -m venv venv && venv/bin/pip install tiktoken==0.14.0
```

```python
import glob, tiktoken
enc = tiktoken.get_encoding("cl100k_base")
def unquote(v):
    v = v.strip()
    if len(v) >= 2 and v[0] == "'" and v[-1] == "'":
        return v[1:-1].replace("''", "'")
    if len(v) >= 2 and v[0] == '"' and v[-1] == '"':
        return v[1:-1]
    return v
total = 0
for p in sorted(glob.glob("agents/*.md")):
    t = open(p, encoding="utf-8").read()
    for line in t[3:t.find("\n---", 3)].splitlines():
        if line.startswith("description:"):
            total += len(enc.encode(unquote(line[12:])))
print(total)  # 2728
```
